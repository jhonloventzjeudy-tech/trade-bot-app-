`; return; } // 2. Pou Crypto (BTC/USDT) - Rale done dirèk nan Binance if (selectedPair === "BTC/USDT") { try { const response = await fetch("https://api.binance.com/api/v3/klines?symbol=BTCUSDT&interval=1m&limit=30"); const rawData = await response.json(); // Konvèti done Binance yo nan fòma bouji (Open, High, Low, Close) const candles = rawData.map(c => ({ time: c[0], open: parseFloat(c[1]), high: parseFloat(c[2]), low: parseFloat(c[3]), close: parseFloat(c[4]) })); // Kalkile endikatè yo ak vrè kòd la const closes = candles.map(c => c.close); const rsi = calculateRSI(closes); const ema9 = calculateEMA(closes, 9); const ema21 = calculateEMA(closes, 21); const atr = calculateATR(candles); const last = candles[candles.length - 1]; const previous = candles[candles.length - 2]; const volatilityPct = (atr / last.close) * 100; let signal = "NO TRADE"; let reason = "Pa gen ase konfimasyon."; let boxClass = "result-box neutral"; if (volatilityPct > 0.5) { reason = "Gwo volatilite detekte! Mache a pa stab."; boxClass = "result-box danger"; } else if (ema9 > ema21 && last.close > ema9 && rsi >= 50 && rsi <= 68) { signal = "CALL (BUY) 🟢"; reason = "Tandans anlè + RSI konfime an tan reyèl!"; boxClass = "result-box buy"; } else if (ema9 < ema21 && last.close < ema9 && rsi >= 32 && rsi <= 50) { signal = "PUT (SELL) 🔴"; reason = "Tandans anba + RSI konfime an tan reyèl!"; boxClass = "result-box sell"; } res.className = boxClass; res.innerHTML = `

RSI: ${rsi ? rsi.toFixed(2) : 'N/A'} | Volatilite: ${volatilityPct.toFixed(3)}%
📊 ${reason}
`; return; } catch (err) { res.className = "result-box danger"; res.textContent = "Echèk nan rale done nan Binance API. Re-eseye."; return; } } // 3. Pou pè Forex reyèl yo (EUR/USD Real) si pa gen backend res.className = "result-box neutral"; res.innerHTML = `
